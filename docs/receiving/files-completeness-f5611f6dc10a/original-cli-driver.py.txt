import json
import os
import subprocess
import sys
import threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from urllib.parse import urlsplit

root = Path(__file__).resolve().parent
source = root / "source"
requests = []
case = "healthy"

class Handler(BaseHTTPRequestHandler):
    def log_message(self, *_):
        pass

    def do_GET(self):
        requests.append(self.path)
        url = urlsplit(self.path)
        status, body, headers = 200, [], {}
        if url.path == "/api/v1/courses/7/files" and not url.query.startswith("cursor="):
            body = [{"id": 1, "display_name": "First course file", "size": 0}]
            headers["Link"] = f'<http://127.0.0.1:{self.server.server_port}/api/v1/courses/7/files?cursor=next>; rel="next"'
        elif url.path == "/api/v1/courses/7/files":
            if case in ("403", "404", "500"):
                status, body = int(case), {"error": "synthetic later-page refusal"}
            elif case == "malformed":
                body = {"unexpected": "object after list"}
            else:
                body = [{"id": 2, "display_name": "Nested course file", "size": None}]
        elif url.path == "/api/v1/courses/7/folders/by_path":
            body = [{"id": 80, "name": "course files"}]
        elif url.path == "/api/v1/folders/80/files":
            body = [{"id": 99, "display_name": "Root-only decoy", "size": 3}]
        else:
            status, body = 404, {"error": "unexpected route"}
        payload = json.dumps(body).encode()
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(payload)))
        for key, value in headers.items():
            self.send_header(key, value)
        self.end_headers()
        self.wfile.write(payload)

server = ThreadingHTTPServer(("127.0.0.1", 0), Handler)
worker = threading.Thread(target=server.serve_forever, daemon=True)
worker.start()
env = {key: value for key, value in os.environ.items() if "proxy" not in key.lower()}
env["PYTHONPATH"] = os.pathsep.join(str(root / name) for name in ("source/src", "deps", "mcp-deps"))
env["PYTHONDONTWRITEBYTECODE"] = "1"
observations = []
try:
    for case in ("healthy", "403", "404", "500", "malformed"):
        requests.clear()
        command = [sys.executable, "-m", "canvaspilot.cli", "files", "7",
                   "--base-url", f"http://127.0.0.1:{server.server_port}",
                   "--token", "synthetic-fixture-token", "--profile", str(root / "unused-profile")]
        result = subprocess.run(command, capture_output=True, text=True, env=env, timeout=20)
        observations.append({"case": case, "returncode": result.returncode,
                             "stdout": result.stdout, "stderr": result.stderr,
                             "requests": requests.copy()})
finally:
    server.shutdown()
    server.server_close()
    worker.join()
Path(sys.argv[1]).write_text(json.dumps(observations, indent=2) + "\n")
for row in observations:
    print(json.dumps({"case": row["case"], "exit": row["returncode"],
                      "ids": [x["id"] for x in json.loads(row["stdout"])] if row["stdout"].strip() else [],
                      "requests": row["requests"]}))
